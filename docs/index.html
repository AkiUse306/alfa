Quick start

Launch Alfa in minutes.

Install the release

# Linux
curl -L -o alfa.tar.gz https://github.com/AkiUse306/alfa/releases/download/1.0/alfa-1.0-linux.pkg
mkdir -p /opt/alfa && tar -xzf alfa.tar.gz -C /opt/alfa

# Windows
powershell -Command "Invoke-WebRequest -Uri 'https://github.com/AkiUse306/alfa/releases/download/1.0/alfa-1.0-windows.msi' -OutFile 'alfa-1.0-windows.msi'"

# macOS
curl -L -o alfa-1.0-macos.pkg https://github.com/AkiUse306/alfa/releases/download/1.0/alfa-1.0-macos.pkg

Start the platform

# Start server
cd /opt/alfa/dist/server && dotnet Alfa.Server.dll

# Start CLI
cd /opt/alfa/dist/cli && dotnet Alfa.Cli.dll --help

# Open web admin
open http://localhost:5000
